cmake_minimum_required(VERSION 3.16)
project(PC98PLAYER CXX RC)
set(CMAKE_CXX_STANDARD 17)
file(GLOB CORE core/*.cpp)
set(SRC ${CORE} win32/main.cpp)
if(MSVC)
  list(APPEND SRC win32/pc98player.rc)
else()
  list(APPEND SRC win32/pc98player_mingw.rc)
endif()
set(YMFM ${CMAKE_SOURCE_DIR}/third_party/ymfm/src)
if(EXISTS ${YMFM}/ymfm_opn.cpp)
  list(APPEND SRC ${YMFM}/ymfm_opn.cpp ${YMFM}/ymfm_adpcm.cpp ${YMFM}/ymfm_ssg.cpp)
  add_compile_definitions(GMPV3_WITH_YMFM)
  include_directories(${YMFM})
endif()
add_executable(PC98PLAYER WIN32 ${SRC})
target_compile_definitions(PC98PLAYER PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS NOMINMAX)
if(MSVC)
  target_compile_options(PC98PLAYER PRIVATE /utf-8)
  set_property(TARGET PC98PLAYER PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_link_options(PC98PLAYER PRIVATE -municode -static)
endif()
target_link_libraries(PC98PLAYER winmm gdi32 user32 shell32 comdlg32 setupapi advapi32 ole32 uuid)
